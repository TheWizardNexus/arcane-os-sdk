cmake_minimum_required(VERSION 3.26)
project(ArcaneKokoroFrontend LANGUAGES CXX)

# The producer builds the disclosed upstream projects through their own build systems.
# This target consumes their public headers/static libraries; it acquires nothing.
set(ESPEAK_NG_INCLUDE_DIRECTORY "" CACHE PATH "Public include directory containing espeak-ng/speak_lib.h")
set(ESPEAK_NG_LIBRARY "" CACHE FILEPATH "Matching static espeak-ng library")
set(UCD_LIBRARY "" CACHE FILEPATH "Matching eSpeak Unicode character database static library")
set(OPUS_INCLUDE_DIRECTORY "" CACHE PATH "Public include directory containing opus.h")
set(OPUS_LIBRARY "" CACHE FILEPATH "Disclosed libopus 1.6.1 static library")
set(OPUSENC_INCLUDE_DIRECTORY "" CACHE PATH "Public include directory containing opusenc.h")
set(OPUSENC_LIBRARY "" CACHE FILEPATH "Disclosed libopusenc 0.3 static library")
set(ESPEAK_NG_DATA_DIRECTORY "" CACHE PATH "Complete generated same-revision espeak-ng-data directory")

add_executable(arcane-kokoro main.cpp)
target_compile_features(arcane-kokoro PRIVATE cxx_std_17)
target_compile_definitions(arcane-kokoro PRIVATE LIBESPEAK_NG_EXPORT=1)
if(WIN32)
    target_compile_definitions(arcane-kokoro PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
target_include_directories(arcane-kokoro PRIVATE
    "${ESPEAK_NG_INCLUDE_DIRECTORY}" "${OPUS_INCLUDE_DIRECTORY}" "${OPUSENC_INCLUDE_DIRECTORY}")
target_link_libraries(arcane-kokoro PRIVATE
    "${ESPEAK_NG_LIBRARY}" "${UCD_LIBRARY}" "${OPUSENC_LIBRARY}" "${OPUS_LIBRARY}")
if(MSVC)
    target_compile_options(arcane-kokoro PRIVATE /utf-8 /EHsc)
elseif(UNIX)
    target_link_libraries(arcane-kokoro PRIVATE m)
elseif(MINGW)
    target_link_options(arcane-kokoro PRIVATE -municode)
endif()
install(TARGETS arcane-kokoro RUNTIME DESTINATION bin)
if(ESPEAK_NG_DATA_DIRECTORY)
    install(DIRECTORY "${ESPEAK_NG_DATA_DIRECTORY}/" DESTINATION share/espeak-ng-data)
endif()

if(WIN32 AND MSVC)
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
    set(CMAKE_INSTALL_DEBUG_LIBRARIES FALSE)
    set(CMAKE_INSTALL_DEBUG_LIBRARIES_ONLY FALSE)
    set(CMAKE_INSTALL_UCRT_LIBRARIES FALSE)
    set(CMAKE_INSTALL_MFC_LIBRARIES FALSE)
    set(CMAKE_INSTALL_OPENMP_LIBRARIES FALSE)
    include(InstallRequiredSystemLibraries)
endif()
